class Solution {
    public boolean isAnagram(String s, String t) {

        if (s.length() != t.length()) {
            return false;
        }

        int[] freqArr = new int[26];

        char[] sArr = s.toCharArray();
        char[] tArr = t.toCharArray();

        for (int i = 0; i < t.length(); i++) {
            freqArr[sArr[i] - 'a']++;
            freqArr[tArr[i] - 'a']--;
        }

        for (int i = 0; i < freqArr.length; i++) {
            if (freqArr[i] != 0) {
                return false;
            }
        }

        return true;
    }
}
