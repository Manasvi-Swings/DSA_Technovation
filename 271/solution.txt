class Solution {
  public:
    string encode(vector<string>& arr) {
        // write your logic to encode the strings
        string encoded_str = "";
                for (const string& s : arr) {
                    encoded_str += to_string(s.length()) + "/:" + s;
                }
                return encoded_str;
    }

    vector<string> decode(string& s) {
        // write your logic to decode the string
        vector<string> decoded_arr;
                int i = 0;
                int n = s.length();

                while (i < n) {
                    
                    size_t delim_idx = s.find("/:", i);
                    if (delim_idx == string::npos) {
                        break;
                    }

                    int length = stoi(s.substr(i, delim_idx - i));
                
                    int start_idx = delim_idx + 2;
                    decoded_arr.push_back(s.substr(start_idx, length));
                    i = start_idx + length;
                }

                return decoded_arr;
    }
};
